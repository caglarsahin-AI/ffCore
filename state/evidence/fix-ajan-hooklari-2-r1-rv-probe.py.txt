def call(label, payload, extra_env=None, raw=None, expect=None):
    env = {k: v for k, v in os.environ.items() if not k.startswith("FFCORE_")}
    env.pop("CLAUDE_CODE_HOST_SESSION_ID", None)
    env["FFCORE_MAIN_ROOT"] = R
    env["FFCORE_GUARD_MODE"] = "block"
    env.update(extra_env or {})
    data = raw if raw is not None else json.dumps(payload, ensure_ascii=False)
    p = subprocess.run(CMD, input=data, capture_output=True, text=True, encoding="utf-8", env=env)
    mark = "" if expect is None else ("  OK" if p.returncode == expect else f"  !! beklenen {expect}")
    results.append((label, p.returncode, expect))
    print(f"## {label}")
    print(f"   input={data[:260]!r}")
    print(f"   exit={p.returncode}{mark}")
    if p.stdout.strip():
        print(f"   stdout={p.stdout.strip()[:400]}")
    for line in p.stderr.strip().splitlines():
        print(f"   stderr={line[:300]}")


def bash(cmd):
    return {"tool_name": "Bash", "tool_input": {"command": cmd}, "session_id": "sess-probe"}


def write(path, content):
    return {"tool_name": "Write", "tool_input": {"file_path": path, "content": content}, "session_id": "sess-probe"}


def edit(path, old, new, every=False):
    return {"tool_name": "Edit", "tool_input": {"file_path": path, "old_string": old, "new_string": new, "replace_all": every}, "session_id": "sess-probe"}


def multi(path, edits):
    return {"tool_name": "MultiEdit", "tool_input": {"file_path": path, "edits": edits}, "session_id": "sess-probe"}
