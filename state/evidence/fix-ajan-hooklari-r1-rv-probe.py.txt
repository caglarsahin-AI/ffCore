def call(label, payload, extra_env=None, raw=None):
    env = {k: v for k, v in os.environ.items() if not k.startswith("FFCORE_")}
    env.pop("CLAUDE_CODE_HOST_SESSION_ID", None)
    env["FFCORE_MAIN_ROOT"] = R
    env.update(extra_env or {})
    data = raw if raw is not None else json.dumps(payload)
    p = subprocess.run(CMD, input=data, capture_output=True, text=True, encoding="utf-8", env=env)
    shown = {k: v for k, v in (extra_env or {}).items()}
    print(f"## {label}")
    print(f"   env={shown} input={data[:200]}")
    print(f"   exit={p.returncode}")
    if p.stdout.strip():
        print(f"   stdout={p.stdout.strip()}")
    if p.stderr.strip():
        for line in p.stderr.strip().splitlines():
            print(f"   stderr={line}")
