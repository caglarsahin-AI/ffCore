const backend = () => process.env.FFCORE_BACKEND_URL!;
const log = (...a: unknown[]) => console.log("[PROBE]", ...a);

async function cookieFor(email: string) {
  const r = await fetch(`${backend()}/api/v1/auth/login`, { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ email, password: PASSWORD }) });
  expect(r.status).toBe(200);
  return (r.headers.get("set-cookie") ?? "").split(";")[0];
}
async function as(email: string, method: string, path: string, body?: unknown) {
  const r = await fetch(`${backend()}/api/v1${path}`, { method, headers: { "content-type": "application/json", cookie: await cookieFor(email) }, body: body === undefined ? undefined : JSON.stringify(body) });
  const text = await r.text();
  let code: string | undefined;
  try { code = JSON.parse(text)?.error?.code; } catch { /* not json */ }
  return { status: r.status, code, text: text.slice(0, 300) };
}
async function member(role: string, access: "edit" | "read") {
  const email = `e2e-probe-${role}-${access}-${Math.random().toString(36).slice(2, 7)}@example.org`;
  const u = await apiCall<{ id: string }>("POST", "/users", { email, full_name: `Probe ${role}`, password: PASSWORD, is_superadmin: false, superadmin_access: null });
  await apiCall("POST", `/projects/${fixture().alpha.id}/members`, { user_id: u.id, role, access });
  return email;
}

test("P2 SSE unknown event + drop: unknown ignored, progress shown, lost after reopens, then GET polling", async ({ page, context, baseURL }) => {
  test.setTimeout(120_000);
  const { alpha, analyst } = fixture();
  const warnings: string[] = [];
  page.on("console", (m) => { if (m.type() === "warning") warnings.push(m.text()); });
  await signIn(page, analyst.email);
  await selectProject(context, baseURL!, alpha.id);
  const real = await asAnalyst<Record<string, unknown>>("GET", `/projects/${alpha.id}/eval-runs/${data.exam.id}`);
  let gets = 0;
  let streams = 0;
  await page.route(`**/api/v1/projects/${alpha.id}/eval-runs/${data.exam.id}`, (route) => {
    gets++;
    return route.fulfill({ status: 200, contentType: "application/json", body: JSON.stringify({ ...real, status: "running", summary: null, scored: 0, items: 3 }) });
  });
  await page.route(`**/eval-runs/${data.exam.id}/stream`, (route) => {
    streams++;
    return route.fulfill({ status: 200, contentType: "text/event-stream", body: 'event: heartbeat\ndata: {"x":1}\n\nevent: progress\ndata: {"status":"running","items":3,"scored":1}\n\n' });
  });
  const t0 = Date.now();
  await page.goto(`/evaluations/runs/${data.exam.id}`);
  await expect(page.getByTestId("eval-progress")).toContainText("1 / 3");
  await expect(page.getByTestId("eval-stream-state")).toHaveAttribute("data-state", "lost", { timeout: 60_000 });
  const lostAfter = Date.now() - t0;
  const getsAtLost = gets;
  await expect(page.getByText(en.Evaluations.exam.streamLostHint)).toBeVisible();
  await page.waitForTimeout(7_000);
  log(`streams opened=${streams} lost after ${lostAfter} ms; GETs at lost=${getsAtLost}, 7 s later=${gets}; warnings=${JSON.stringify(warnings.filter((w) => w.includes("eval stream")))}`);
  expect(streams).toBe(6);
  expect(gets).toBeGreaterThan(getsAtLost);
  expect(warnings.some((w) => w.includes('unknown event "heartbeat"'))).toBe(true);
});

test("P3 live: backend killed during a waiting exam (host kills and starts it on the marker)", async ({ page, context, baseURL }) => {
  test.setTimeout(420_000);
  const { alpha, analyst } = fixture();
  const hello = await helloAgent(alpha.id, "probe-live");
  const { agent } = await approvalAgent(alpha.id);
  const versions = await asAnalyst<{ items: { id: string; version: number }[] }>("GET", `/projects/${alpha.id}/agents/${agent.id}/versions`);
  const approvalVersion = versions.items.find((v) => v.version === 1)!.id;
  // No gated agent; approve_tools on: the item's run waits in the approval queue, so the exam stays running.
  const suite = await createSuite(alpha.id, "probe-live", { approve_tools: true });
  await addItem(alpha.id, suite.id, hello.runId);
  await signIn(page, analyst.email);
  await selectProject(context, baseURL!, alpha.id);
  let exam: { id: string } | null = null;
  await expect.poll(async () => {
    try { exam = await startExam(alpha.id, suite.id, approvalVersion); return "started"; } catch (e) { if (String(e).includes("dataset_empty")) return "waiting"; throw e; }
  }, { timeout: 30_000, intervals: [1000] }).toBe("started");
  const id = exam!.id;
  await page.goto(`/evaluations/runs/${id}`);
  await expect(page.getByTestId("eval-stream-state")).toHaveAttribute("data-state", "streaming");
  await expect(page.getByTestId("eval-status")).toHaveText(/Running|Queued/);
  writeFileSync("/work/.review-probe-marker", `${Date.now()}\n`);
  log("marker written; exam", id);
  const seen: string[] = [];
  const t0 = Date.now();
  const snap = async () => {
    const state = await page.getByTestId("eval-stream-state").getAttribute("data-state").catch(() => "?");
    const prog = await page.getByTestId("eval-progress").innerText().catch(() => "?");
    const status = await page.getByTestId("eval-status").innerText().catch(() => "?");
    const err = await page.locator("[role=alert]").count().catch(() => -1);
    const line = `${state} | ${prog} | ${status} | alerts=${err}`;
    if (seen[seen.length - 1]?.split(" @")[0] !== line) seen.push(`${line} @${Math.round((Date.now() - t0) / 1000)}s`);
  };
  // Until the backend is back and the page has said lost.
  let back = false;
  for (;;) {
    await snap();
    try { const r = await fetch(`${process.env.FFCORE_BACKEND_URL}/healthz`); back = r.status < 500; } catch { back = false; }
    const state = await page.getByTestId("eval-stream-state").getAttribute("data-state").catch(() => "?");
    if (back && state === "lost" && Date.now() - t0 > 30_000) break;
    if (Date.now() - t0 > 180_000) break;
    await page.waitForTimeout(1000);
  }
  for (let i = 0; i < 8; i++) { await snap(); await page.waitForTimeout(1000); }
  log("sequence before cancel:", JSON.stringify(seen));
  await page.getByTestId("eval-cancel").click();
  for (let i = 0; i < 90; i++) {
    await snap();
    if ((await page.getByTestId("eval-status").innerText().catch(() => "")) === "Cancelled") break;
    await page.waitForTimeout(1000);
  }
  log("sequence after cancel:", JSON.stringify(seen.slice(-6)));
  await expect(page.getByTestId("eval-status")).toHaveText("Cancelled");
});

test("P7 keyboard only: create a suite with Tab/Space/Enter", async ({ page, context, baseURL }) => {
  const { alpha, analyst } = fixture();
  await signIn(page, analyst.email);
  await selectProject(context, baseURL!, alpha.id);
  await page.goto("/evaluations");
  await page.getByTestId("primary-action").focus();
  await page.keyboard.press("Enter");
  const tabTo = async (pred: string) => {
    for (let i = 0; i < 60; i++) {
      await page.keyboard.press("Tab");
      if (await page.evaluate((p) => { const a = document.activeElement as HTMLElement | null; return !!a && (a.id === p || a.getAttribute("data-testid") === p); }, pred)) return;
    }
    throw new Error(`could not tab to ${pred}`);
  };
  const key = uniqueKey("kb").slice(0, 60);
  await page.locator("body").focus();
  await tabTo("suite-key"); await page.keyboard.type(key);
  await tabTo("suite-name"); await page.keyboard.type("Keyboard suite");
  await tabTo("scorer-contains"); await page.keyboard.press("Space");
  await tabTo("suite-threshold"); await page.keyboard.type("0.5");
  await tabTo("suite-save"); await page.keyboard.press("Enter");
  await expect(page.getByTestId("suite-title")).toHaveText("Keyboard suite");
  log("keyboard-only suite created", key);
});
