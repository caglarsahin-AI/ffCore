// DIAG (isci-c 10c-1 r2): not part of the PR; untracked, deleted after the run. Reads a live exam's
// stream and logs when each chunk arrives: through the console's proxy with and without gzip, and
// from the backend directly.
import { expect, test } from "@playwright/test";

import { approvalAgent } from "./agents";
import { addItem, asAnalyst, createSuite, helloAgent, startExam } from "./eval";
import { fixture } from "./fixtures";
import { selectProject, signIn } from "./support";

async function timing(url: string, cookie: string, encoding: string) {
  const t0 = Date.now();
  const lines: string[] = [];
  const controller = new AbortController();
  const stop = setTimeout(() => controller.abort(), 12_000);
  try {
    const r = await fetch(url, { headers: { cookie, "accept-encoding": encoding }, signal: controller.signal });
    lines.push(`status ${r.status} enc=${r.headers.get("content-encoding")} @${((Date.now() - t0) / 1000).toFixed(1)}s`);
    const reader = r.body!.getReader();
    for (;;) {
      const { done, value } = await reader.read();
      if (done) break;
      lines.push(`chunk ${value.length}B @${((Date.now() - t0) / 1000).toFixed(1)}s`);
    }
  } catch (e) {
    lines.push(`end ${String(e).slice(0, 40)} @${((Date.now() - t0) / 1000).toFixed(1)}s`);
  } finally {
    clearTimeout(stop);
  }
  return lines;
}

test("diag: chunk timing of a live exam stream, proxy gzip / proxy identity / backend", async ({ page, context, baseURL }) => {
  test.setTimeout(240_000);
  const { alpha, analyst } = fixture();
  const hello = await helloAgent(alpha.id, "diag");
  const { agent } = await approvalAgent(alpha.id);
  const versions = await asAnalyst<{ items: { id: string; version: number }[] }>("GET", `/projects/${alpha.id}/agents/${agent.id}/versions`);
  const suite = await createSuite(alpha.id, "diag", { approve_tools: true });
  await addItem(alpha.id, suite.id, hello.runId);
  let id = "";
  await expect(async () => {
    id = (await startExam(alpha.id, suite.id, versions.items[0].id)).id;
  }).toPass({ timeout: 30_000 });
  await signIn(page, analyst.email);
  await selectProject(context, baseURL!, alpha.id);
  const cookie = (await context.cookies()).map((c) => `${c.name}=${c.value}`).join("; ");
  const path = `/api/v1/projects/${alpha.id}/eval-runs/${id}/stream`;
  console.log("[DIAG] proxy gzip", JSON.stringify(await timing(`${baseURL}${path}`, cookie, "gzip, deflate, br")));
  console.log("[DIAG] proxy identity", JSON.stringify(await timing(`${baseURL}${path}`, cookie, "identity")));
  console.log("[DIAG] backend", JSON.stringify(await timing(`${process.env.FFCORE_BACKEND_URL}${path}`, cookie, "gzip, deflate, br")));
  await asAnalyst("POST", `/projects/${alpha.id}/eval-runs/${id}/cancel`);
});
